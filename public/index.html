Como Conectar seu Bot na VPS / Host (Python)
# Script Python para rodar na sua VPS (Long Polling):
import requests, time

TOKEN = "SEU_TOKEN_AQUI"
BASE = f"http://localhost:3000/api/bot{TOKEN}"
offset = 0

print(f"🤖 Bot iniciado com sucesso!")

while True:
    try:
        r = requests.get(f"{BASE}/getUpdates?offset={offset}&timeout=20").json()
        for update in r.get("result", []):
            offset = update["update_id"] + 1
            msg = update.get("message", {})
            chat_id = msg.get("chat", {}).get("id")
            text = msg.get("text", "")
            user = msg.get("from", {}).get("first_name", "Alguém")
            
            print(f"[{user}]: {text}")
            
            # Resposta customizada do seu bot:
            if text == "/ping":
                requests.post(f"{BASE}/sendMessage", json={
                    "chat_id": chat_id,
                    "text": f"Pong! Olá {user}, bot 100% online!"
                })
    except Exception as e:
        time.sleep(2)
